Property Viewer Demo

Read-only companion to Property Editor for details panes and record summaries.

createPropertyViewer(host, data, {
  showCopyButtons: false, // default for rows without copyable
  onCopy(payload) {
    writeLog("viewer.copy", payload);
  },
  onAction(property, action, meta) {
    writeLog("viewer.action", { property, action, meta });
  },
});
