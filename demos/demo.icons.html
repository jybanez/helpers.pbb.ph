Curated Icon Pack

This catalog is generated from the live registry, so new shared icons appear here automatically as the pack expands.

import { uiLoader } from "../js/ui/ui.loader.js?v=0.21.187";
import { FILE_ICONS, getFileIconName } from "../js/ui/ui.icons.files.js";
import { SOCIAL_ICONS } from "../js/ui/ui.icons.social.js";
import { AI_ICONS } from "../js/ui/ui.icons.ai.js";
await uiLoader.load("ui.icons");
const { createIcon, registerIconPack } = await uiLoader.get("ui.icons");
// Import/register only the packs your app needs.
registerIconPack(FILE_ICONS);
registerIconPack(SOCIAL_ICONS);
registerIconPack(AI_ICONS);
const attachment = createIcon(getFileIconName("report.pdf"), { title: "PDF" });
const github = createIcon("social.github", { title: "GitHub" });
const claude = createIcon("ai.claude", { variant: "brand", title: "Claude" });
const unknown = createIcon("ai.future-provider", { fallback: "ai.generic" });
document.getElementById("your-icon-host").append(attachment, github, claude, unknown);
