Composition Sample

      
<div class="workspace">
  <div id="threeColumnHost"></div>
  <aside class="fixedColumn">Column 4</aside>
</div>

<style>
.workspace {
  display: grid;
  grid-template-columns: minmax(0, 1fr) 220px;
  gap: 10px;
  min-height: 340px;
}
</style>

const nestedHost = document.createElement("div");
nestedHost.style.cssText = "height: 100%; min-width: 0; min-height: 0";

createSplitter(threeColumnHost, {
  panePadding: 0,
  orientation: "horizontal",
  initialRatio: 0.28,
  minRatio: 0.18,
  maxRatio: 0.5,
  paneA: column1,
  paneB: nestedHost,
});

createSplitter(nestedHost, {
  panePadding: 0, chrome: false,
  orientation: "horizontal",
  initialRatio: 0.48,
  minRatio: 0.25,
  maxRatio: 0.75,
  paneA: column2,
  paneB: column3,
});
